class Solution {
    public int largestAltitude(int[] gain) {
        int altitude=0;
        int maxaltitude=0;
        for(int i=0;i<gain.length;i++){
            altitude+=gain[i];
            if(altitude>maxaltitude){
                maxaltitude=altitude;
            }
        }
        return maxaltitude;
    }
}


Input
gain =
[-4,-3,-2,-1,4,3,2]
Output
0
